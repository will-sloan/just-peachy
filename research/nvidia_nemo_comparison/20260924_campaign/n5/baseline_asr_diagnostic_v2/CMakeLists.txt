cmake_minimum_required(VERSION 3.20)
project(baseline_asr_diagnostic_v2 LANGUAGES CXX)
if(NOT WIN32 OR NOT DEFINED SHERPA_ROOT)
  message(FATAL_ERROR "Windows build requires explicit verified SHERPA_ROOT")
endif()
add_executable(baseline_asr_diagnostic_v2 diagnostic.cpp)
target_compile_features(baseline_asr_diagnostic_v2 PRIVATE cxx_std_17)
target_compile_options(baseline_asr_diagnostic_v2 PRIVATE /W4 /utf-8)
target_include_directories(baseline_asr_diagnostic_v2 PRIVATE "${SHERPA_ROOT}/include")
target_link_libraries(baseline_asr_diagnostic_v2 PRIVATE "${SHERPA_ROOT}/lib/sherpa-onnx-c-api.lib")
